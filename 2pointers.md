def max_number(nums1, nums2, k):

    # Get the largest subsequence of length k
    def max_subsequence(nums, k):
        stack = []
        remove = len(nums) - k

        for num in nums:
            # Remove smaller digits if we can
            while stack and remove and stack[-1] < num:
                stack.pop()
                remove -= 1

            stack.append(num)

        return stack[:k]

    # Merge two subsequences into the largest possible number
    def merge(a, b):
        result = []
        i = j = 0

        while i < len(a) or j < len(b):

            if a[i:] > b[j:]:
                result.append(a[i])
                i += 1
            else:
                result.append(b[j])
                j += 1

        return result

    best = []

    # x = number of digits taken from nums1
    for x in range(max(0, k - len(nums2)), min(k, len(nums1)) + 1):

        a = max_subsequence(nums1, x)
        b = max_subsequence(nums2, k - x)
